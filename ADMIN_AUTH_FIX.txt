const adminAuthentication = async () => {
  console.log("adminAuthentication")
  try {
    const response = await fetch(`${API_BASE_URL}/auth/login`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({"email":"admin@spa.com","password":"Admin@123"})
    });
    
    if (!response.ok) {
      console.error("Admin authentication failed:", response.status);
      return temp_token;
    }
    
    const data = await response.json();
    console.log("Admin auth data:", data);
    const token = data.token || data.data?.token;
    console.log("Admin auth token:", token);

    console.error("Admin authentication error:", error);
    return temp_token;
  }
}
